var wySw4 = new Swiper("#wySw4", {
  loop: true,
  speed: 800,
  effect: 'cube',
  autoplay: {
    delay: 3200,
    disableOnInteraction: false,   // ✅ 补齐：手动后不停

    el: "#wyPag4",
    clickable: true,               // ✅ 补齐：圆点可点

🎬 任务二 · creative 电影墙（超纲 API 仿写）

原始需求
为演示四新增第 7 个按钮「creative 电影墙」：
- 上一张（prev）缩小、左移、带角度离场
- 下一张（next）放大、右移进场
- 参数自定，但要"像电影选片墙"
最终代码（swiper-task.html:330-346）
creativeEffect: {
  prev: {                           // 上一张离场
    translate: ['-30%', 0, 0],      // 左移容器宽 30%
    scale: 0.85,                    // 缩小到 85%
    rotate: [-15, 0, 'deg'],       // 逆时针转 15°
    opacity: 0.45,                  // 半透明
  },
  next: {                           // 下一张进场
    translate: ['30%', 0, 0],       // 右移容器宽 30%
    scale: 1.15,                    // 放大到 115%
    rotate: [15, 0, 'deg'],        // 顺时针转 15°
    opacity: 0.45,                  // 半透明
  },
}
📖 答辩追问准备：
• prev 控制什么？ 当前展示 slide 的"前一张兄弟"离场动画 — 在左边，逐渐变小、向左飞出、带逆时针角度
• next 控制什么？ 当前展示 slide 的"后一张兄弟"进场动画 — 在右边，逐渐放大、向右飞入、带顺时针角度
• 百分号相对谁？ 相对 Swiper 容器本身 的宽度（不是屏幕）。translate: ['-30%', 0, 0] = X 轴偏移容器宽的 30%
• 改数值让卡片转 20°？ 把两个 -15 / 15 改成 -20 / 20
